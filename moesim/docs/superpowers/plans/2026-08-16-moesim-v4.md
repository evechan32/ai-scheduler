#!/usr/bin/env bash
set -euo pipefail

# moesim 一键安装脚本
# 用法:
#   ./install.sh            # 核心 (numpy only, 模拟器+调度器)
#   ./install.sh --full     # + torch/transformers/accelerate (执行层)
#   ./install.sh --gpu-llama# + CUDA 版 llama-cpp-python (真机推理)
#   ./install.sh --dev      # + pytest (开发)

VENV_DIR="${VENV_DIR:-.venv}"
PYTHON="${PYTHON:-python3}"

# 1. 检测 Python 版本
if ! command -v "$PYTHON" >/dev/null 2>&1; then
  echo "错误: 未找到 python3" >&2; exit 1
fi
PY_VER=$("$PYTHON" -c 'import sys; print(f"{sys.version_info.major}.{sys.version_info.minor}")')
echo "[1/5] Python: $PY_VER"
if [[ "$(echo "$PY_VER" | cut -d. -f1)" -lt 3 || ("$(echo "$PY_VER" | cut -d. -f1)" -eq 3 && "$(echo "$PY_VER" | cut -d. -f2)" -lt 10) ]]; then
  echo "错误: 需要 Python >= 3.10 (当前 $PY_VER)" >&2; exit 1
fi

# 2. 创建 venv
if [[ ! -d "$VENV_DIR" ]]; then
  echo "[2/5] 创建 venv at $VENV_DIR"
  "$PYTHON" -m venv "$VENV_DIR"
fi
source "$VENV_DIR/bin/activate"
echo "[2/5] venv 就绪: $("$PYTHON" -c 'import sys; print(sys.executable)')"

# 3. 安装核心依赖
echo "[3/5] 安装核心依赖 (numpy)"
pip install -q --upgrade pip
pip install -q -e ".[core]"

# 4. 可选组件
for arg in "$@"; do
  case "$arg" in
    --full)
      echo "[4/5] 安装执行层 (torch/transformers/accelerate)"
      pip install -q -e ".[executor]"
      ;;
    --gpu-llama)
      echo "[4/5] 编译 CUDA 版 llama-cpp-python (需 nvidia-smi)"
      if ! command -v nvidia-smi >/dev/null 2>&1; then
        echo "警告: 未检测到 CUDA, 跳过 GPU llama.cpp" >&2
      else
        CMAKE_ARGS="-DGGML_CUDA=on" pip install -q --force-reinstall --no-cache-dir llama-cpp-python
      fi
      ;;
    --dev)
      echo "[4/5] 安装开发依赖 (pytest)"
      pip install -q -e ".[dev]"
      ;;
  esac
done

# 5. 验证
echo "[5/5] 运行测试验证"
if python -c "import pytest" >/dev/null 2>&1; then
  python -m pytest tests/ -q
else
  python -c "from moesim.sim.sweep import compare_policies; print('核心模块 OK (numpy-only)')"
fi
echo "✅ 安装完成"

### Task 31: 调度决策优化（决策缓存，缩小与 HF 原生差距）

**Files:**
- Modify: `moesim/executor/backends/forward_hook.py`
- Modify: `moesim/executor/backends/transformers.py`
- Test: `tests/executor/test_forward_hook.py`（追加）

**Interfaces:**
- 现状：hook 每层每 forward 都构建 ScheduleState + 调用 scheduler.decide——这是混合执行 29.47ms vs HF 8.30ms 的主要开销。
- 增强：`MoEForwardHook` 增加 `decision_cache: dict[tuple[int, tuple[str,...]], tuple[set[str], ...]]`——缓存 (layer_idx, requested) → 决策（哪些专家 CPU）。`requested` 相同且 `residency` 未变化时复用缓存决策，跳过 decide。
- 测试：`test_cache_reuses_decisions`——同层同 requested 两次调用，scheduler.decide 只调用 1 次（用 RecordingScheduler）。

- [ ] **Step 1: 写失败测试**（追加 test_cache_reuses_decisions 到 test_forward_hook.py）

- [ ] **Step 3: 实现缓存**

在 `MoEForwardHook.__init__` 增加 `self.decision_cache: dict = {}`；`_make_forward` 的 forward 里：
```python

if cache_key in self.decision_cache:
    cpu_experts = self.decision_cache[cache_key]
else:

```
同时在 `_forward_3d`/`_forward_2d` 用缓存结果。residency 变化时（load/unload）应使缓存失效——v4 简化：缓存按 (layer, requested) 键，且 residency 变化由 executor.load 触发时清空缓存（在 executor.load/unload 里 hook 需感知——通过 `MoEForwardHook` 提供 `invalidate()`，executor 不感知，改为：缓存只在同一 forward 序列内有效，跨 forward 失效——最简单：每次 forward 开始时清空缓存）。

**关键设计选择**：缓存只在**同一次 forward 调用内**有效（每层只 decide 一次，多层相同 requested 复用），跨 forward 自动失效（保证 correctness）。测试里同层连续两次 mlp(x) 调用是两次独立 forward——这样计数还是 2！

**修正测试设计**：真正的优化是**同一 forward 内多 token 共享决策**（当前每 token 每层都 decide）。改为：一次 forward 只 decide 一次/层，缓存 key=(layer_idx, requested) 在**单次 forward 内**跨 token 复用。测试改为：构造 2 层模型，两层 requested 相同 → decide 调用次数 == 层数（而非 token×层）。或者更实际：缓存按 (layer_idx, requested) 且 forward 生命周期内有效，测试断言多 token 单层只 decide 1 次。

**实现建议**（implementer 判断）：在 hook 增加 `_in_forward` 标记，forward 开始置 True、结束置 False；缓存仅在 `_in_forward` 期间有效，退出即清。这样：
- 单层多 token：decide 1 次 ✓
- 多层相同 requested：decide 每层 1 次（不能跨层复用，因为专家模块不同）
测试断言：2 层模型、每层 8 专家、单 forward 多 token，decide 总调用 == 2（每层 1 次）——对比优化前 == token×层。

**以 implementer 验证的最终测试为准**，核心断言必须是"decide 调用次数显著减少且正确性不变"。

def expert_ffn_int4_gemm(hidden_states, w1, w2, scale_act: float = 1.0) -> torch.Tensor:
    """INT4 weights with real int8 gemm: dequant-packed->int8, activate->int8, torch._int_mm.

    Accuracy-first; quantization of activations is per-tensor with scale chosen
    to keep the 10% relative-error bound.
    """
    q1, s1 = _quantize_int4(w1)   # packed uint8, 2-per-byte
    q2, s2 = _quantize_int4(w2)
    # unpack to int8 (values -8..7)
    lo = (q1.to(torch.int16) & 0xF) - 8
    hi = ((q1.to(torch.int16) >> 4) & 0xF) - 8
    w1_i8 = torch.stack([lo, hi], dim=-1).reshape(-1)[: w1.numel()].reshape(w1.shape).to(torch.int8)
    lo = (q2.to(torch.int16) & 0xF) - 8
    hi = ((q2.to(torch.int16) >> 4) & 0xF) - 8
    w2_i8 = torch.stack([lo, hi], dim=-1).reshape(-1)[: w2.numel()].reshape(w2.shape).to(torch.int8)

    # quantize activations to int8
    x = hidden_states.float()
    x_amax = x.abs().max().clamp(min=1e-6)
    x_scale = x_amax / 127.0
    x_i8 = torch.clamp(torch.round(x / x_scale), -127, 127).to(torch.int8)

    h = torch._int_mm(x_i8, w1_i8.t()).float() * (x_scale * s1)
    h = torch.nn.functional.gelu(h)
    # second matmul: h int8
    h_amax = h.abs().max().clamp(min=1e-6)
    h_scale = h_amax / 127.0
    h_i8 = torch.clamp(torch.round(h / h_scale), -127, 127).to(torch.int8)
    out = torch._int_mm(h_i8, w2_i8.t()).float() * (h_scale * s2)
    return out.to(hidden_states.dtype)
```

class RLPolicyGradientScheduler(Scheduler):
    """REINFORCE-style policy gradient over expert placement (numpy-only).

    theta[expert, action] with actions {0: load, 1: execute_cpu}.
    Softmax policy; returns scalar reward -sum(cpu_ms) per step; gradient ascent.
    """

    def __init__(self, n_experts: int, alpha: float = 0.05, gamma: float = 0.9) -> None:
        self.n_experts = n_experts
        self.alpha = alpha
        self.gamma = gamma
        self.theta = np.zeros((n_experts, 2), dtype=np.float64)

    def _probs(self, expert_id: int) -> np.ndarray:
        logits = self.theta[expert_id]
        e = np.exp(logits - logits.max())
        return e / e.sum()

    def _sample_action(self, expert_id: int) -> str:
        p = self._probs(expert_id)
        return "load" if np.random.rand() < p[0] else "execute_cpu"

    def decide(self, state, clock):
        actions = []
        for eid in state.requested:
            if eid in state.resident:
                actions.append(Action(kind="execute_gpu", expert_ids=(eid,)))
            else:
                actions.append(Action(kind=self._sample_action(int(eid)), expert_ids=(eid,)))
        return actions

    def train(self, sim, episodes=100, trace=None):
        trace = trace or [["e0", "e1"]]
        for _ in range(episodes):
            for step in trace:
                sim._state.requested = tuple(step)
                actions = self.decide(sim._state, sim._clock)
                reward = -sum(
                    sim.profiles[eid].cpu_exec_ms
                    for a in actions if a.kind == "execute_cpu" for eid in a.expert_ids
                )
                for a in actions:
                    if a.kind in ("load", "execute_cpu"):
                        i = int(a.expert_ids[0])
                        a_idx = 0 if a.kind == "load" else 1
                        p = self._probs(i)
                        # REINFORCE: grad = reward * (onehot - prob)
                        grad = reward * (1.0 - p[a_idx])
                        self.theta[i, a_idx] += self.alpha * grad
```
